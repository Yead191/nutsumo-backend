2. CLIENT-SIDE HELPER UTILITY (Add to utils/taxCalculator.ts)
====================================================================================================
Please create or update a helper function in the frontend for live fee calculation (e.g. when an artisan
types a quotation amount or customer views price breakdown):

```typescript
export const TAX_RATES = {
  VAT: 15,            // 15%
  NHIL: 2.5,          // 2.5%
  GET_FUND_LEVY: 2.5, // 2.5%
} as const;

export const TOTAL_DEDUCTION_PERCENTAGE = 20;

export interface IFeeBreakdown {
  grossAmount: number;
  vatAmount: number;
  nhilAmount: number;
  getFundLevyAmount: number;
  totalDeductions: number;
  netPayoutAmount: number;
}

export const calculateFeeBreakdown = (amount: number): IFeeBreakdown => {
  const safeAmount = Number(amount) || 0;
  const vatAmount = Number(((safeAmount * TAX_RATES.VAT) / 100).toFixed(2));
  const nhilAmount = Number(((safeAmount * TAX_RATES.NHIL) / 100).toFixed(2));
  const getFundLevyAmount = Number(((safeAmount * TAX_RATES.GET_FUND_LEVY) / 100).toFixed(2));
  const totalDeductions = Number((vatAmount + nhilAmount + getFundLevyAmount).toFixed(2));
  const netPayoutAmount = Number((safeAmount - totalDeductions).toFixed(2));

  return {
    grossAmount: safeAmount,
    vatAmount,
    nhilAmount,
    getFundLevyAmount,
    totalDeductions,
    netPayoutAmount,
  };
};
```
